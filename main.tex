\subsubsection{Design Rationale}

\example{\textbf{Example:} The architecture is designed to achieve a balance between predictive performance and computational efficiency. The feature extraction backbone is responsible for capturing discriminative representations from the input data, whereas the feature fusion module enhances the representation of objects or patterns appearing at different scales. This design is especially useful in practical scenarios where target samples vary in size, illumination, viewpoint, and background complexity. A lightweight prediction head is adopted to reduce the number of trainable parameters and improve inference speed. Therefore, the overall architecture is expected to provide a practical trade-off among accuracy, model complexity, and deployment feasibility.}

\noindent\guide{Hướng dẫn: Cần giải thích vì sao thiết kế như vậy. Ví dụ: dùng multi-scale feature fusion vì đối tượng có nhiều kích thước; dùng attention vì cần tập trung vào vùng quan trọng; dùng mô hình nhẹ vì cần FPS cao; dùng augmentation vì dữ liệu ít; dùng loss đặc biệt vì mất cân bằng lớp hoặc cần định vị chính xác hơn. Thiết kế nào cũng cần có lý do, không nên chỉ liệt kê module.}

\example{\textbf{Example:} The input--output mapping of the model can be generally formulated as:}
\begin{equation}
    \hat{y} = f_{\theta}(x),
\end{equation}
\example{where $x$ is the input sample, $f_{\theta}(\cdot)$ denotes the model with learnable parameters $\theta$, and $\hat{y}$ is the predicted output. For localization or detection tasks, the model predicts a set of regions, labels, and confidence scores:}
\begin{equation}
    \hat{Y} = \{(b_i, c_i, s_i)\}_{i=1}^{N},
\end{equation}
\example{where $b_i$ is the predicted region, $c_i$ is the predicted class, $s_i$ is the confidence score, and $N$ is the number of predicted objects or regions.}

\subsubsection{Example of Loss Function and Optimization Objective}

\example{\textbf{Example:} The model is trained by minimizing a task-specific objective function. For a classification task, the cross-entropy loss can be used to measure the discrepancy between the predicted probability distribution and the ground-truth label:}
\begin{equation}
    \mathcal{L}_{cls} = -\sum_{k=1}^{K} y_k \log(\hat{y}_k),
\end{equation}
\example{where $K$ denotes the number of classes, $y_k$ is the ground-truth indicator for class $k$, and $\hat{y}_k$ is the predicted probability. For a detection task, the overall loss can be expressed as a weighted combination of classification, localization, and confidence losses:}
\begin{equation}
    \mathcal{L}_{total} = \lambda_{cls}\mathcal{L}_{cls} + \lambda_{box}\mathcal{L}_{box} + \lambda_{obj}\mathcal{L}_{obj},
\end{equation}
\example{where $\mathcal{L}_{box}$ penalizes localization errors, $\mathcal{L}_{obj}$ measures objectness or confidence errors, and $\lambda_{cls}$, $\lambda_{box}$, and $\lambda_{obj}$ are balancing coefficients.}

\noindent\guide{Hướng dẫn: Phải nêu rõ hàm loss dùng trong mô hình. Nếu dùng mô hình có sẵn, cần tìm hiểu loss gốc hoặc loss được framework sử dụng. Nếu loss gồm nhiều thành phần, cần giải thích mỗi thành phần tối ưu điều gì. Mô tả loss sơ sài hoặc không đúng với bài toán sẽ bị trừ điểm.}

\subsubsection{Example of Training and Inference Algorithm}

\example{\textbf{Example:} Algorithm~\ref{alg:model_training_inference} summarizes the training and inference procedure. During training, the model receives mini-batches of input samples and corresponding labels, computes predictions, evaluates the loss, and updates the learnable parameters through gradient-based optimization. During inference, the trained model processes unseen test samples and produces final predictions after task-specific post-processing.}

\begin{algorithm}[H]
\caption{Example training and inference procedure}
\label{alg:model_training_inference}
\small
\begin{enumerate}[leftmargin=*, itemsep=2pt]
    \item \textbf{Input:} Training set $\mathcal{D}_{\mathrm{train}}$, validation set $\mathcal{D}_{\mathrm{val}}$, model $f_{\Theta}$, learning rate $\eta$, and number of epochs $T$.
    \item \textbf{Output:} Trained model parameters $\Theta^{*}$ and final predictions $\hat{Y}$.
    \item Initialize model parameters $\Theta$.
    \item For each epoch $t = 1, 2, \ldots, T$, repeat the following steps:
    \begin{enumerate}[label*=\arabic*., leftmargin=*, itemsep=1pt]
        \item Load each mini-batch $(x, y)$ from $\mathcal{D}_{\mathrm{train}}$.
        \item Compute prediction $\hat{y} = f_{\Theta}(x)$.
        \item Compute training loss $\mathcal{L}(y, \hat{y})$.
        \item Update parameters using gradient descent: $\Theta \leftarrow \Theta - \eta \nabla_{\Theta}\mathcal{L}$.
    \end{enumerate}
    \item Evaluate the model on $\mathcal{D}_{\mathrm{val}}$ and save the best checkpoint.
    \item Apply the best trained model to the test set.
    \item Perform post-processing if required, such as thresholding, non-maximum suppression, voting, filtering, or tracking.
    \item Return final predictions $\hat{Y}$.
\end{enumerate}
\end{algorithm}

Nếu bài toán có hậu xử lý như NMS, thresholding, tracking, voting, ranking hoặc filtering thì cần đưa vào thuật toán. \subsection{Evaluation Metrics}

\example{The models are evaluated using both predictive-performance metrics and computational-efficiency metrics. Predictive metrics include precision, recall, F1-score, accuracy, mAP, IoU, or Dice score depending on the task. Efficiency metrics include number of parameters, model size, FLOPs, FPS, and inference time.}

\begin{equation}
    \text{Precision} = \frac{TP}{TP + FP}, \quad
    \text{Recall} = \frac{TP}{TP + FN},
\end{equation}
\begin{equation}
    F1 = 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}.
\end{equation}
\begin{equation}
    \text{IoU} = \frac{|P \cap G|}{|P \cup G|}, \quad
    \text{DSC} = \frac{2|P \cap G|}{|P| + |G|}.
\end{equation}

\noindent\guide{Hướng dẫn: Chọn chỉ số phù hợp với bài toán. Bài toán phân loại nên có accuracy, precision, recall, F1-score và confusion matrix. Bài toán phát hiện nên có mAP, precision, recall, Params, FLOPs và FPS. Bài toán phân đoạn nên có IoU, Dice, precision, recall, Params, FLOPs và FPS nếu phù hợp.}
